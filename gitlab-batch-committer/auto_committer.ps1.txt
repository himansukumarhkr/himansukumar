function Pause-ForUser {
    Write-Host ""
    try {
        $null = [System.Console]::ReadKey($true)
    } catch {
        cmd.exe /c pause
    }
}

function Protect-Pat($plainText) {
    Add-Type -AssemblyName System.Security
    $bytes = [System.Text.Encoding]::UTF8.GetBytes($plainText)
    $enc = [System.Security.Cryptography.ProtectedData]::Protect($bytes, $null, [System.Security.Cryptography.DataProtectionScope]::LocalMachine)
    return [Convert]::ToBase64String($enc)
}

function Unprotect-Pat($base64Text) {
    Add-Type -AssemblyName System.Security
    $bytes = [Convert]::FromBase64String($base64Text)
    $dec = [System.Security.Cryptography.ProtectedData]::Unprotect($bytes, $null, [System.Security.Cryptography.DataProtectionScope]::LocalMachine)
    return [System.Text.Encoding]::UTF8.GetString($dec)
}

$ScriptDir   = $PSScriptRoot
$ConfigFile  = Join-Path $ScriptDir "config.toon"
$SourceFile  = Join-Path $ScriptDir "source.txt"

$sourceCreatedFirstTime = $false
if (-not (Test-Path $SourceFile)) {
    Set-Content -Path $SourceFile -Value "" -Encoding UTF8
    Write-Host "[INFO] Created blank 'source.txt'." -ForegroundColor Yellow
    $sourceCreatedFirstTime = $true
}

function Read-ConfigFile($path) {
    if (-not (Test-Path $path)) { return $null }
    $state = @{}
    $compulsory = ""
    $phrases = [System.Collections.Generic.List[string]]::new()
    $suffixes = [System.Collections.Generic.List[string]]::new()
    $currentSection = ""

    foreach ($line in (Get-Content -Path $path -Encoding UTF8)) {
        $trimmed = $line.Trim()
        if ([string]::IsNullOrWhiteSpace($trimmed) -or $trimmed.StartsWith("#")) {
            continue
        }

        if ($trimmed -match '^\[(.*)\]$') {
            $currentSection = $Matches[1].ToUpper()
            continue
        }

        switch ($currentSection) {
            "STATE" {
                if ($trimmed -match '^([^=]+)=(.*)$') {
                    $key = $Matches[1].Trim()
                    $val = $Matches[2].Trim()
                    $state[$key] = $val
                }
            }
            "COMPULSORY" {
                if ([string]::IsNullOrWhiteSpace($compulsory)) {
                    $compulsory = $trimmed
                }
            }
            "PHRASES" {
                $phrases.Add($trimmed)
            }
            { $_ -in "SUFFIX", "SUFFIXES", "COAUTHORS", "COAUTHOR", "POSTFIX", "RANDOM_COMPULSORY" } {
                $suffixes.Add($trimmed)
            }
            default {
                $phrases.Add($trimmed)
            }
        }
    }

    return @{
        State      = $state
        Compulsory = $compulsory
        Phrases    = @($phrases)
        Suffixes   = @($suffixes)
    }
}

function Save-ConfigFile($path, $configData) {
    $state = $configData.State
    $sb = [System.Text.StringBuilder]::new()
    
    [void]$sb.AppendLine("[STATE]")
    [void]$sb.AppendLine("targetFile = $($state['targetFile'])")
    [void]$sb.AppendLine("numCommits = $($state['numCommits'])")
    [void]$sb.AppendLine("linesPerCommit = $($state['linesPerCommit'])")
    [void]$sb.AppendLine("cursor = $($state['cursor'])")
    [void]$sb.AppendLine("pat = $($state['pat'])")
    [void]$sb.AppendLine("")
    
    [void]$sb.AppendLine("[COMPULSORY]")
    [void]$sb.AppendLine($configData.Compulsory)
    [void]$sb.AppendLine("")

    [void]$sb.AppendLine("[PHRASES]")
    foreach ($p in $configData.Phrases) {
        [void]$sb.AppendLine($p)
    }
    [void]$sb.AppendLine("")

    [void]$sb.AppendLine("[COAUTHORS]")
    foreach ($c in $configData.Suffixes) {
        [void]$sb.AppendLine($c)
    }

    Set-Content -Path $path -Value $sb.ToString().TrimEnd() -Encoding UTF8
}

function Get-FormattedCommitMessage($configData) {
    $phrases = $configData.Phrases
    $selectedPhrase = if ($phrases -and $phrases.Count -gt 0) {
        $phrases | Get-Random
    } else {
        "update component definitions"
    }

    $selectedSuffix = ""
    if ($configData.Suffixes -and $configData.Suffixes.Count -gt 0) {
        $pick = $configData.Suffixes | Get-Random
        if ($pick.Trim().StartsWith("Co-Authored-By:", [System.StringComparison]::OrdinalIgnoreCase)) {
            $selectedSuffix = $pick.Trim()
        } elseif (-not [string]::IsNullOrWhiteSpace($pick) -and $pick.Trim().ToLower() -ne "none") {
            $selectedSuffix = $pick.Trim()
        }
    }

    $baseMsg = if (-not [string]::IsNullOrWhiteSpace($configData.Compulsory)) {
        $comp = $configData.Compulsory.Trim()
        if ($comp -match '[:\- \]\)]\s*$') {
            "$comp $selectedPhrase"
        } else {
            "$comp - $selectedPhrase"
        }
    } else {
        $selectedPhrase
    }

    if (-not [string]::IsNullOrWhiteSpace($selectedSuffix)) {
        return "$baseMsg $selectedSuffix"
    } else {
        return $baseMsg
    }
}

function New-StateConfig {
    Write-Host "`n[SETUP] Configuring GitLab Batch Committer..." -ForegroundColor Cyan
    $patInput  = Read-Host "Enter your GitLab Personal Access Token (PAT)"
    $targetUrl = Read-Host "Enter Target GitLab File URL"
    
    $numCommits = Read-Host "Enter number of commits per run [Default: 5]"
    $numCommits = if ([string]::IsNullOrWhiteSpace($numCommits)) { 5 } else { [int]$numCommits }

    $linesPer = Read-Host "Enter lines per commit [Default: 2]"
    $linesPer = if ([string]::IsNullOrWhiteSpace($linesPer)) { 2 } else { [int]$linesPer }

    $encryptedPat = Protect-Pat $patInput
    Write-Host "[SUCCESS] PAT securely encrypted with LocalMachine binding." -ForegroundColor Green

    $defaultPhrases = @(
        "refactor core logic",
        "enhance stability and fix edge cases",
        "optimize performance metrics",
        "update execution handlers and fixtures",
        "add validation for dynamic payload",
        "clean up redundant helper methods"
    )

    $defaultCoauthors = @(
        "Co-Authored-By: Duo",
        "Co-Authored-By: Copilot",
        "Co-Authored-By: None"
    )

    $newConfig = @{
        State = @{
            targetFile     = $targetUrl
            numCommits     = $numCommits
            linesPerCommit = $linesPer
            cursor         = 0
            pat            = $encryptedPat
        }
        Compulsory = "BVB-24160"
        Phrases    = $defaultPhrases
        Suffixes   = $defaultCoauthors
    }

    Save-ConfigFile -path $ConfigFile -configData $newConfig
    Write-Host "[SUCCESS] Configuration saved to config.toon!`n" -ForegroundColor Green
    return $newConfig
}

$isFirstRun = $false
$configData = Read-ConfigFile $ConfigFile

if ($null -eq $configData -or [string]::IsNullOrWhiteSpace($configData.State["pat"]) -or [string]::IsNullOrWhiteSpace($configData.State["targetFile"])) {
    Write-Host "[INFO] Valid config.toon not found. Running initial setup..." -ForegroundColor Cyan
    $configData = New-StateConfig
    $isFirstRun = $true
}

if ($isFirstRun -or $sourceCreatedFirstTime) {
    Write-Host "===========================================================" -ForegroundColor Green
    Write-Host "[SETUP COMPLETE] Initial configuration saved to 'config.toon'." -ForegroundColor Green
    Write-Host "[ACTION REQUIRED] Paste your source code into 'source.txt'" -ForegroundColor Yellow
    Write-Host "and run the script again to begin committing." -ForegroundColor Yellow
    Write-Host "===========================================================`n" -ForegroundColor Green
    Pause-ForUser
    return
}

$rawSource = if (Test-Path $SourceFile) { Get-Content -Path $SourceFile -Raw -Encoding UTF8 } else { "" }
if ([string]::IsNullOrWhiteSpace($rawSource)) {
    Write-Host "`n[WARNING] 'source.txt' is empty." -ForegroundColor Yellow
    Write-Host "Please paste your source code into 'source.txt' and re-run to start committing.`n" -ForegroundColor Cyan
    Pause-ForUser
    return
}

$pat = ""
try {
    $pat = Unprotect-Pat $configData.State["pat"]
} catch {
    Write-Host "[ERROR] Failed to decrypt PAT. This file is locked to its original machine." -ForegroundColor Red
    Pause-ForUser
    return
}

function Parse-GitLabUrl($url, $token) {
    $cleanUrl = if ($url -match '^([^?]+)') { $Matches[1] } else { $url }

    if ($cleanUrl -notmatch '/-/blob/') {
        throw "Invalid GitLab URL format: Missing '/-/blob/' separator in $url"
    }

    $parts = $cleanUrl -split '/-/blob/'
    $leftPart  = $parts[0]
    $rightPart = $parts[1]

    if ($leftPart -match '^(https?://[^/]+)/(.*?)$') {
        $domain      = $Matches[1]
        $projectPath = $Matches[2]
    } else {
        throw "Could not parse domain and project path from: $leftPart"
    }

    $rightSegments = $rightPart -split '/', 2
    $branch   = $rightSegments[0]
    $filePath = $rightSegments[1]

    $encodedPath = [System.Uri]::EscapeDataString($projectPath)
    $lookupUri   = "$($domain)/api/v4/projects/$($encodedPath)"

    Write-Host "[INFO] Resolving numeric Project ID for '$projectPath'..." -ForegroundColor Cyan
    try {
        $proj = Invoke-RestMethod -Uri $lookupUri -Headers @{ "PRIVATE-TOKEN" = $token } -Method Get -UseDefaultCredentials -ErrorAction Stop
        $numericId = $proj.id
        Write-Host "[INFO] Successfully connected! Project: $($proj.name) (ID: $numericId)" -ForegroundColor Green
    } catch {
        throw "Failed to resolve Project ID from GitLab API. Verify PAT and permissions: $_"
    }

    $decodedFilePath = [System.Net.WebUtility]::UrlDecode($filePath)
    $fileEncoded = ($decodedFilePath -split '/' | ForEach-Object { [System.Uri]::EscapeDataString($_) }) -join '%2F'

    $apiUrl = "$($domain)/api/v4/projects/$($numericId)/repository/files/$($fileEncoded)"

    return @{
        ApiUrl   = $apiUrl
        Branch   = $branch
        FilePath = $decodedFilePath
    }
}

try {
    $targetInfo = Parse-GitLabUrl $configData.State["targetFile"] $pat
} catch {
    Write-Host "[WARNING] Connection check failed: $_" -ForegroundColor Yellow
    $configData = New-StateConfig
    $pat = Unprotect-Pat $configData.State["pat"]
    $targetInfo = Parse-GitLabUrl $configData.State["targetFile"] $pat
}

$headers = @{ "PRIVATE-TOKEN" = $pat }

$fullContent = $rawSource
$fullContent = $fullContent -replace "`r`n", "`n" -replace "`r", "`n"
$lines = $fullContent -split "`n"
$totalLines = $lines.Count
$cursor = [int]$configData.State["cursor"]

$phraseCount = if ($configData.Phrases) { $configData.Phrases.Count } else { 0 }
$suffixCount = if ($configData.Suffixes) { $configData.Suffixes.Count } else { 0 }
Write-Host "[INFO] Loaded config.toon ($phraseCount phrases | $suffixCount co-authors | Compulsory: '$($configData.Compulsory)')" -ForegroundColor Cyan
Write-Host "[INFO] Total source lines: $totalLines | Starting Cursor: $cursor" -ForegroundColor Cyan

if ($cursor -ge $totalLines) {
    Write-Host "[INFO] Cursor reached end of file. Resetting to 0." -ForegroundColor Yellow
    $cursor = 0
}

function Get-TargetFileContent($apiUrl, $branch, $headers) {
    $uri = "$($apiUrl)?ref=$($branch)"
    try {
        $res = Invoke-RestMethod -Uri $uri -Headers $headers -Method Get -UseDefaultCredentials -ErrorAction Stop
        $decoded = [System.Convert]::FromBase64String($res.content)
        return [System.Text.Encoding]::UTF8.GetString($decoded)
    } catch {
        $status = $_.Exception.Response.StatusCode.value__
        if ($status -eq 404) { return "" }
        if ($status -eq 401 -or $status -eq 403) { throw "AUTH_ERROR" }
        return $null
    }
}

function Commit-TargetFileContent($apiUrl, $branch, $content, $message, $headers) {
    $body = @{
        branch         = $branch
        commit_message = $message
        content        = $content
        encoding       = "text"
    } | ConvertTo-Json

    $checkUri = "$($apiUrl)?ref=$($branch)"
    $exists = $false
    try {
        $null = Invoke-RestMethod -Uri $checkUri -Headers $headers -Method Get -UseDefaultCredentials -ErrorAction Stop
        $exists = $true
    } catch {
        $exists = $false
    }

    $method = if ($exists) { "Put" } else { "Post" }
    try {
        $null = Invoke-RestMethod -Uri $apiUrl -Headers $headers -Method $method -Body $body -ContentType "application/json" -UseDefaultCredentials -ErrorAction Stop
        return $true
    } catch {
        Write-Error "[API ERROR] Failed to push commit ($method): $_"
        return $false
    }
}

Write-Host "[INFO] Fetching current content from GitLab..." -ForegroundColor Cyan
$existingContent = Get-TargetFileContent -apiUrl $targetInfo.ApiUrl -branch $targetInfo.Branch -headers $headers
if ($null -eq $existingContent) {
    $existingContent = ""
}

$numCommits = [int]$configData.State["numCommits"]
$linesPer   = [int]$configData.State["linesPerCommit"]

for ($i = 1; $i -le $numCommits; $i++) {
    if ($cursor -ge $totalLines) {
        Write-Host "[INFO] Reached the end of source.txt. Ending run." -ForegroundColor Yellow
        break
    }

    $endIndex   = [Math]::Min($cursor + $linesPer, $totalLines)
    $batchLines = $lines[$cursor..($endIndex - 1)]
    $batchText  = $batchLines -join "`n"

    Write-Host "`n----------------------------------------" -ForegroundColor DarkGray
    Write-Host "Batch Step $i of $numCommits (Lines: $($cursor + 1) to $endIndex):" -ForegroundColor Magenta
    foreach ($line in $batchLines) {
        Write-Host "  > $line" -ForegroundColor Gray
    }

    if ([string]::IsNullOrEmpty($existingContent)) {
        $existingContent = $batchText
    } else {
        $existingContent += "`n" + $batchText
    }

    $msg = Get-FormattedCommitMessage -configData $configData

    Write-Host "Commit Message: '$msg'" -ForegroundColor DarkCyan
    $ok = Commit-TargetFileContent -apiUrl $targetInfo.ApiUrl -branch $targetInfo.Branch -content $existingContent -message $msg -headers $headers

    if ($ok) {
        $cursor = $endIndex
        $configData.State["cursor"] = $cursor
        Save-ConfigFile -path $ConfigFile -configData $configData
        Write-Host "[SUCCESS] Commit pushed successfully!" -ForegroundColor Green
    } else {
        Write-Host "[FAILED] Commit failed. Aborting remaining batches." -ForegroundColor Red
        Pause-ForUser
        break
    }
}

Write-Host "`n[DONE] Batch run completed. Current saved cursor: $cursor" -ForegroundColor Cyan
Pause-ForUser
