@echo off
setlocal
cd /d "%~dp0"

echo =======================================================
echo          GitLab Batch Committer Launcher
echo =======================================================
echo.

if exist "%~dp0auto_committer.ps1" (
    powershell.exe -NoExit -NoProfile -ExecutionPolicy Bypass -File "%~dp0auto_committer.ps1"
) else (
    powershell.exe -NoExit -NoProfile -ExecutionPolicy Bypass -File "%~dp0auto_committer.ps1.txt"
)

if %ERRORLEVEL% neq 0 (
    echo.
    echo [ERROR] The process completed with an exit code of %ERRORLEVEL%.
)

echo.
pause
